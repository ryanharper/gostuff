// 3. Drop RDP/Interactive user sessions and Idle/System noise
otelcol.processor.filter "drop_user_sessions" {
  error_mode = "ignore"

  metrics {
    datapoint = [
      "IsMatch(resource.attributes[\"process.executable.name\"], \"^(?i)(Idle|System|rdpclip\\\\.exe|explorer\\\\.exe|taskhostw\\\\.exe|sihost\\\\.exe|ctfmon\\\\.exe|dwm\\\\.exe|RuntimeBroker\\\\.exe|SearchHost\\\\.exe|StartMenuExperienceHost\\\\.exe)$\")",
    ]
  }

  output {
    metrics = [otelcol.processor.transform.resolve_names.input]
  }
}

// 4. Step A of Transform: Resolve command_line to service/instance names & strip PIDs
otelcol.processor.transform "resolve_names" {
  error_mode = "ignore"

  metric_statements {
    context = "resource"
    statements = [
      // Default to executable name without .exe
      "set(attributes[\"process\"], replace_pattern(attributes[\"process.executable.name\"], \"(?i)\\\\.exe$\", \"\"))",

      // Resolve svchost: "-s <ServiceName>" (or fallback "-k <GroupName>")
      "set(attributes[\"process\"], Concat([\"svchost\", ExtractPatterns(attributes[\"process.command_line\"], \"(?i)-k\\\\s+(?P<grp>[A-Za-z0-9_]+)\")[\"grp\"]], \"_\")) where IsMatch(attributes[\"process.executable.name\"], \"(?i)^svchost\") and IsMatch(attributes[\"process.command_line\"], \"(?i)-k\\\\s+[A-Za-z0-9_]+\")",
      "set(attributes[\"process\"], Concat([\"svchost\", ExtractPatterns(attributes[\"process.command_line\"], \"(?i)-s\\\\s+(?P<svc>[A-Za-z0-9_-]+)\")[\"svc\"]], \"_\")) where IsMatch(attributes[\"process.executable.name\"], \"(?i)^svchost\") and IsMatch(attributes[\"process.command_line\"], \"(?i)-s\\\\s+[A-Za-z0-9_-]+\")",

      // Resolve sqlservr: "-s<InstanceName>" or "-s <InstanceName>"
      "set(attributes[\"process\"], Concat([\"sqlservr\", ExtractPatterns(attributes[\"process.command_line\"], \"(?i)-s\\\\s*(?P<inst>[A-Za-z0-9_\\\\$]+)\")[\"inst\"]], \"_\")) where IsMatch(attributes[\"process.executable.name\"], \"(?i)^sqlservr\") and IsMatch(attributes[\"process.command_line\"], \"(?i)-s\\\\s*[A-Za-z0-9_\\\\$]+\")",

      // Resolve sqlagent: "-i <InstanceName>"
      "set(attributes[\"process\"], Concat([\"sqlagent\", ExtractPatterns(attributes[\"process.command_line\"], \"(?i)-i\\\\s+(?P<inst>[A-Za-z0-9_\\\\$]+)\")[\"inst\"]], \"_\")) where IsMatch(attributes[\"process.executable.name\"], \"(?i)^sqlagent\") and IsMatch(attributes[\"process.command_line\"], \"(?i)-i\\\\s+[A-Za-z0-9_\\\\$]+\")",

      // Sanitize any '$' in SQL instance names to '_'
      "replace_pattern(attributes[\"process\"], \"\\\\$\", \"_\")",

      // Strip all high-cardinality PID and command line resource attributes
      "delete_key(attributes, \"process.pid\")",
      "delete_key(attributes, \"process.parent_pid\")",
      "delete_key(attributes, \"process.command_line\")",
      "delete_key(attributes, \"process.executable.path\")",
      "delete_key(attributes, \"process.executable.name\")",
      "delete_key(attributes, \"process.owner\")",
      "delete_key(attributes, \"process.creation.time\")",
    ]
  }

  output {
    metrics = [otelcol.processor.groupbyattrs.merge_by_process.input]
  }
}

// 5. Merge separate process Resources that now share the same "process" name (e.g. 8 WmiPrvSE instances)
otelcol.processor.groupbyattrs "merge_by_process" {
  keys = ["process"]

  output {
    metrics = [otelcol.processor.transform.rename_to_windows_exporter.input]
  }
}

// 6. Step B of Transform: Rename metrics/labels to windows_process_* and sum merged datapoints
otelcol.processor.transform "rename_to_windows_exporter" {
  error_mode = "ignore"

  metric_statements {
    context = "datapoint"
    statements = [
      // Copy resolved "process" name onto every metric datapoint
      "set(attributes[\"process\"], resource.attributes[\"process\"])",

      // Map CPU "state" -> "mode" (and rename "system" -> "privileged" to match windows_exporter)
      "set(attributes[\"mode\"], \"privileged\") where attributes[\"state\"] == \"system\"",
      "set(attributes[\"mode\"], \"user\") where attributes[\"state\"] == \"user\"",
      "delete_key(attributes, \"state\")",

      // Map Disk I/O "direction" -> "mode" ("read" / "write")
      "set(attributes[\"mode\"], attributes[\"direction\"]) where attributes[\"direction\"] != nil",
      "delete_key(attributes, \"direction\")",

      // Clear page fault "type" ("major"/"minor") so total page faults sum cleanly into one series
      "delete_key(attributes, \"type\")",
    ]
  }

  metric_statements {
    context = "metric"
    statements = [
      // Rename OTel process.* metrics to match windows_exporter names
      // Clear unit on counters/gauges where windows_exporter omits unit suffixes or uses custom naming
      "set(name, \"windows_process_cpu_time_total\") where name == \"process.cpu.time\"",
      "set(unit, \"\") where name == \"windows_process_cpu_time_total\"",

      "set(name, \"windows_process_working_set_bytes\") where name == \"process.memory.usage\"",
      "set(unit, \"\") where name == \"windows_process_working_set_bytes\"",

      "set(name, \"windows_process_virtual_bytes\") where name == \"process.memory.virtual\"",
      "set(unit, \"\") where name == \"windows_process_virtual_bytes\"",

      "set(name, \"windows_process_io_bytes_total\") where name == \"process.disk.io\"",
      "set(unit, \"\") where name == \"windows_process_io_bytes_total\"",

      "set(name, \"windows_process_io_operations_total\") where name == \"process.disk.operations\"",
      "set(unit, \"\") where name == \"windows_process_io_operations_total\"",

      "set(name, \"windows_process_handles\") where name == \"process.handles\"",
      "set(unit, \"\") where name == \"windows_process_handles\"",

      "set(name, \"windows_process_threads\") where name == \"process.threads\"",
      "set(unit, \"\") where name == \"windows_process_threads\"",

      "set(name, \"windows_process_page_faults_total\") where name == \"process.paging.faults\"",
      "set(unit, \"\") where name == \"windows_process_page_faults_total\"",

      // Finally, sum datapoints that share the same process & mode labels (collapses WmiPrvSE, conhost, dllhost)
      "aggregate_metric_labels(\"sum\")",
    ]
  }

  output {
    metrics = [otelcol.processor.filter.drop_cpu_wait.input]
  }
}

// 7. Drop CPU "wait" datapoints (OTel emits state="wait", which has no mode attribute after step 6)
otelcol.processor.filter "drop_cpu_wait" {
  error_mode = "ignore"

  metrics {
    datapoint = [
      "metric.name == \"windows_process_cpu_time_total\" and attributes[\"mode\"] == nil",
    ]
  }

  output {
    metrics = [otelcol.exporter.prometheus.default.input]
  }
}

    $SvcMap[[uint32]$_.ProcessId] += @($_.Name)
}

# 3. Processes where multiple instances should be summed into a single series if not tied to a unique service
$CollapseAlways = @('WmiPrvSE', 'conhost', 'dllhost', 'RuntimeBroker', 'backgroundTaskHost')

# 4. Query raw PerfProc counters and filter to Session 0 (excluding _Total, Idle, and PID 0)
$RawProcs = Get-CimInstance Win32_PerfRawData_PerfProc_Process \vert{} Where-Object {$_.IDProcess -gt 0 -and
    $_.Name -notmatch '^(_Total\vert{}Idle)$' -and
    $Session0Pids.ContainsKey([uint32]$_.IDProcess)
}

$Grouped =$RawProcs | Group-Object {
    $PidVal   = [uint32]$_.IDProcess

    $BaseName = ($_.Name -replace '#\d+$', '')

    if ($BaseName -in @('WmiPrvSE', 'conhost')) {
        # Always sum ephemeral WMI providers and console hosts into one series per binary
        $BaseName

            "${BaseName}_${SvcNames}"
        } else {
            $SvcNames
        }
    }
    else {
        # Any non-service Session 0 process (e.g., multiple worker processes or unmapped dllhost)
        # collapses cleanly by BaseName so it can never cause duplicate series
        $BaseName
    }
}

$Session0Pids = @{}
Get-CimInstance Win32_Process -Filter "SessionId = 0" -Property ProcessId, Name | ForEach-Object {
    $Session0Pids[[uint32]$_.ProcessId] =$_.Name
}

# 2. Map running Windows Services by PID (for svchost, sqlservr, sqlagent, dllhost)
$SvcMap = @{}
Get-CimInstance Win32_Service -Filter "State='Running'" -Property ProcessId, Name | Where-Object ProcessId -gt 0 | ForEach-Object {
    $SvcMap[[uint32]$_.ProcessId] += @($_.Name)
}

# 3. Query PerfProc counters for Session 0 processes only (excluding _Total, Idle, System, and PID 0)
$RawProcs = Get-CimInstance Win32_PerfRawData_PerfProc_Process \vert{} Where-Object {$_.IDProcess -gt 0 -and
    $_.Name -notmatch '^(?i)(_Total\vert{}Idle\vert{}System)$' -and
    $Session0Pids.ContainsKey([uint32]$_.IDProcess)
}

# 4. Group by deterministic process name:
#    - svchost, sqlservr, sqlagent, dllhost -> split by "<BaseName>_<ServiceName>"
#    - wmiapsrv, WmiPrvSE, conhost, and all other Session 0 processes -> collapsed by BaseName
$Grouped =$RawProcs | Group-Object {
    $PidVal   = [uint32]$_.IDProcess
    # Strip the #1, #2 instance suffix that Windows PerfProc appends to duplicate binaries
    $BaseName = ($_.Name -replace '#\d+$', '')

    if ($BaseName -match '^(?i)(svchost\vert{}sqlservr\vert{}sqlagent\vert{}dllhost)$') {
        $Prefix =$BaseName.ToLower()
        if ($SvcMap.ContainsKey($PidVal)) {$SvcNames = ($SvcMap[$PidVal] | Sort-Object) -join "_"
            "${Prefix}_${SvcNames}"
        } else {
            "${Prefix}_unmapped"
        }
    }
    else {
        # Collapses multiple wmiapsrv, WmiPrvSE, conhost, TiWorker, etc. into one unique group per binary
        $BaseName
    }
}
